% Full-width hero (Figure 1, docs/paper/money.png, docs/paper/figures/hero.pdf).
% Two Isosyll marks (v4.2.0 final, live 160 ms) against all 12 timed public CPU comparators.
% x = idle-subtracted core energy per audio-second (log); y = dev-clean WER on the 400-utt
% clean split of the 730-utterance DEV panel (zoomed; systems above \HeroYmax are listed off-scale).
% Only Isosyll is drawn as a large diamond with a bold label; comparators are normal-size marks
% coloured by single-core throughput, with a colour bar.
% Every coordinate comes from data/money_layout.tex.
% Diamonds are nodes at axis coordinates. Do not add point offsets in axis space on this
% log axis: pgfplots displaces those marks and they land on other systems.
\def\herokicker#1#2#3#4{%
  \begin{minipage}[t]{#1}%
  \sffamily\fontsize{8}{9.6}\selectfont\strut\textcolor{#2}{\bfseries #3}%
  \hspace{0.5em}\textcolor{sub}{#4}\par
  \vspace{1.6pt}{\color{#2}\hrule height 0.75pt}%
  \end{minipage}%
}
% \syscolor{Key} defines ecKey (throughput colour) and ncKey (that colour darkened for text).
\def\syscolor#1{%
  \definecolor{ec#1}{HTML}{\csname Hero#1Color\endcsname}%
  \ifnum\csname Hero#1Metered\endcsname>0\relax\colorlet{nc#1}{ec#1!70!black}\else\colorlet{nc#1}{sub}\fi}
% Comparators: normal-size circle (streaming) or square (offline), filled by throughput. Never a diamond.
\def\sysmark#1#2#3{% key, shape (circle|square), unused
  \ifnum\csname Hero#1On\endcsname>0\relax
  \syscolor{#1}%
  \ifx#2c\node[circle,inner sep=0pt,minimum size=5.4pt,fill=ec#1,draw=ink!80,line width=.45pt]
    at (axis cs:\csname Hero#1X\endcsname,\csname Hero#1Y\endcsname) {};\else
  \node[rectangle,inner sep=0pt,minimum size=5pt,fill=ec#1,draw=ink!80,line width=.45pt]
    at (axis cs:\csname Hero#1X\endcsname,\csname Hero#1Y\endcsname) {};\fi
  \fi}
\def\syslabel#1#2#3#4#5{\ifnum\csname Hero#1On\endcsname>0\relax\syscolor{#1}\node[comp,anchor=#2,text=nc#1] at (axis cs:#3,#4) {#5};\fi}
\def\sysleader#1#2{\ifnum\csname Hero#1On\endcsname>0\relax\draw[leader] (axis cs:\csname Hero#1X\endcsname,\csname Hero#1Y\endcsname) #2;\fi}
% Diamond at a log-axis coordinate: a node at (axis cs:...), not a shifted scope.
\def\ourdiamond#1{% key (Final or LiveLm)
  \syscolor{#1}%
  \node[diamond,aspect=1,inner sep=0pt,minimum size=13.6pt,fill=ec#1,draw=ink,line width=.8pt,line join=round]
    at (axis cs:\csname Hero#1X\endcsname,\csname Hero#1Y\endcsname) {};}
