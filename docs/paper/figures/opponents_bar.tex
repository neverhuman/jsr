% Two-column IEEE comparison bar chart (runtime-lane): 1-core Zen 2 CPU Speed (RTFx),
% Pooled DEV Panel WER (%), and Live MFA-aligned Stable-Word Lag (p50 ms).
% Reads ONLY the shared Schema v2 table data/cold_dev.csv (zero duplicate evidence paths);
% automatically filters to headline Zen 2 1-core plotted rows (machine=zen2, threads=1, plot=1)
% and visually separates ULTRA (blue), live streaming comparators (orange), and whole-utterance
% systems (hatched grey).
\input{figures/style.tex}
\pgfplotstableread[col sep=comma]{\paperdatadir cold_dev.csv}\opprawtable
% Assign sequential y-index (0, 1, ...) and per-series y-columns for plotted Zen 2 1-core rows.
\gdef\oppylabels{}%
\newcount\oppcnt\oppcnt=0
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thisplot{\thisrow{plot}}%
    \edef\thismach{\thisrow{machine}}%
    \edef\thisthr{\thisrow{threads}}%
    \def\one{1}\def\zen{zen2}%
    \ifx\thisplot\one\ifx\thismach\zen\ifx\thisthr\one
      \edef\entry{\the\oppcnt}%
      \ifnum\oppcnt>0\xdef\oppylabels{\oppylabels,}\fi
      \xdef\oppylabels{\oppylabels\thisrow{short}}%
      \global\advance\oppcnt by 1
    \else\def\entry{nan}\fi\else\def\entry{nan}\fi\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{baridx}{\opprawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-ours-streaming}%
    \ifx\thiss\want\edef\entry{\thisrow{baridx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{y_ours}{\opprawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-comparator-streaming}%
    \ifx\thiss\want\edef\entry{\thisrow{baridx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{y_stream}{\opprawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-comparator-whole}%
    \ifx\thiss\want\edef\entry{\thisrow{baridx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{y_whole}{\opprawtable}
\pgfmathtruncatemacro{\opplast}{\the\oppcnt-1}
\begin{tikzpicture}
\begin{groupplot}[
  group style={
    group size=3 by 1,
    horizontal sep=16pt,
    yticklabels at=edge left,
  },
  ultra/base,
  clip=true,
  xbar,
  /tikz/bar width=6.8pt,
  height=5.6cm,
  y dir=reverse,
  ymin=-0.65, ymax=\opplast+1.15,
  ytick={0,...,\opplast},
  yticklabels/.expanded={\oppylabels},
  yticklabel style={font=\scriptsize, align=right, inner sep=3pt},
  ymajorgrids=false,
  xmajorgrids=true,
  enlarge y limits=false,
  nodes near coords style={font=\tiny, inner sep=1.8pt, anchor=west},
  label style={font=\scriptsize},
  title style={font=\scriptsize\bfseries, yshift=-2pt},
]

% ---------------- Panel (a): 1-Core CPU Speed (RTFx, log, higher is better)
\nextgroupplot[
  width=4.65cm,
  xmode=log, xmin=0.5, xmax=110, log basis x=10,
  log origin x=infty,
  xtick={1,3,10,30}, xticklabels={1,3,10,30},
  xlabel={1-Core Speed (RTFx $\uparrow$, log)},
  title={(a) 1-Core Speed (Zen 2)},
]
\draw[densely dashed, okVermillion!80, thin] (axis cs:1,-0.65) -- (axis cs:1,\opplast+1.15);
\node[font=\tiny, text=okVermillion, anchor=west, inner sep=1pt]
  at (axis cs:1.12,\opplast+0.82) {1$\times$ RT};
\addplot[fill=okBlue!82, draw=okBlue!95!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}$\times$},
  point meta=explicit]
  table[x=speed, y=y_ours, meta=speed]{\opprawtable};
\addplot[fill=okOrange!80, draw=okOrange!80!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}$\times$},
  point meta=explicit]
  table[x=speed, y=y_stream, meta=speed]{\opprawtable};
\ifnum\ColdRowsWhole>0
\addplot[fill=okGrey!55, draw=okGrey!90!black, postaction={pattern=north east lines, pattern color=white!70}, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}$\times$},
  point meta=explicit]
  table[x=speed, y=y_whole, meta=speed]{\opprawtable};
\fi

% ---------------- Panel (b): Pooled DEV Panel WER (%, lower is better)
\nextgroupplot[
  width=4.45cm,
  xmin=0, xmax=23,
  xtick={0,5,10,15,20}, xticklabels={0,5,10,15,20},
  xlabel={Pooled DEV WER (\% $\downarrow$)},
  title={(b) Pooled DEV WER},
]
\addplot[fill=okBlue!82, draw=okBlue!95!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}\%},
  point meta=explicit]
  table[x=wer, y=y_ours, meta=wer]{\opprawtable};
\addplot[fill=okOrange!80, draw=okOrange!80!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}\%},
  point meta=explicit]
  table[x=wer, y=y_stream, meta=wer]{\opprawtable};
\ifnum\ColdRowsWhole>0
\addplot[fill=okGrey!55, draw=okGrey!90!black, postaction={pattern=north east lines, pattern color=white!70}, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=1]{\pgfplotspointmeta}\%},
  point meta=explicit]
  table[x=wer, y=y_whole, meta=wer]{\opprawtable};
\fi

% ---------------- Panel (c): Live MFA Stable-Word Lag p50 (ms, log, lower is better)
\nextgroupplot[
  width=4.75cm,
  xmode=log, xmin=120, xmax=20000, log basis x=10,
  log origin x=infty,
  xtick={200,500,1000,3000}, xticklabels={0.2,0.5,1,3},
  xlabel={Live Word Lag p50 (s $\downarrow$, log)},
  title={(c) Live MFA Word Lag},
  legend to name=opponentslegend,
  legend columns=3,
  legend style={font=\scriptsize, draw=none, fill=none, /tikz/every even column/.append style={column sep=12pt}},
]
\draw[densely dashed, okGreen!80!black, thin] (axis cs:500,-0.65) -- (axis cs:500,\opplast+1.15);
\node[font=\tiny, text=okGreen!80!black, anchor=west, inner sep=1pt]
  at (axis cs:560,\opplast+0.82) {0.5\,s};
\addplot[fill=okBlue!82, draw=okBlue!95!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,precision=0]{\pgfplotspointmeta}\,ms},
  point meta=explicit]
  table[x=lag_p50_ms, y=y_ours, meta=lag_p50_ms]{\opprawtable};
\addlegendentry{\textsc{Ultra} (streaming, ours)}
\addplot[fill=okOrange!80, draw=okOrange!80!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,precision=0]{\pgfplotspointmeta}\,ms},
  point meta=explicit]
  table[x=lag_p50_ms, y=y_stream, meta=lag_p50_ms]{\opprawtable};
\addlegendentry{Streaming comparators}
\ifnum\ColdRowsWhole>0
\addplot[fill=okGrey!55, draw=okGrey!90!black, postaction={pattern=north east lines, pattern color=white!70}, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,precision=0]{\pgfplotspointmeta}\,ms},
  point meta=explicit]
  table[x=final_latency_p50_ms, y=y_whole, meta=final_latency_p50_ms]{\opprawtable};
\addlegendentry{Whole-utterance (wait + infer)}
\fi
\end{groupplot}
\node[anchor=north] at ($(group c2r1.south)+(0,-0.88cm)$) {\pgfplotslegendfromname{opponentslegend}};
\end{tikzpicture}
