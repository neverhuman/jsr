% Single-column IEEE hardware energy bar chart (runtime-lane): Core energy per audio-hour
% (Wh/audio-h == W per live 1x real-time stream) from AMD Zen 2 MSR CorWatt (idle-bracketed).
% Fits strictly within one IEEE column (<= 252 pt). Reads ONLY shared data/cold_dev.csv.
\input{figures/style.tex}
\pgfplotstableread[col sep=comma]{\paperdatadir cold_dev.csv}\enrawtable
% Assign sequential y-index (0, 1, ...) and per-series y-columns for plotted Zen 2 1-core rows with watts.
\gdef\enylabels{}%
\newcount\encnt\encnt=0
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thisplot{\thisrow{plot}}%
    \edef\thismach{\thisrow{machine}}%
    \edef\thisthr{\thisrow{threads}}%
    \edef\thisw{\thisrow{watts}}%
    \def\one{1}\def\zen{zen2}\def\empty{}%
    \ifx\thisplot\one\ifx\thismach\zen\ifx\thisthr\one\ifx\thisw\empty
      \def\entry{nan}%
    \else
      \edef\entry{\the\encnt}%
      \ifnum\encnt>0\xdef\enylabels{\enylabels,}\fi
      \xdef\enylabels{\enylabels\thisrow{short}}%
      \global\advance\encnt by 1
    \fi\else\def\entry{nan}\fi\else\def\entry{nan}\fi\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{enidx}{\enrawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-ours-streaming}%
    \ifx\thiss\want\edef\entry{\thisrow{enidx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{ey_ours}{\enrawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-comparator-streaming}%
    \ifx\thiss\want\edef\entry{\thisrow{enidx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{ey_stream}{\enrawtable}
\pgfplotstablecreatecol[
  create col/assign/.code={%
    \edef\thiss{\thisrow{series}}\def\want{zen2-t1-comparator-whole}%
    \ifx\thiss\want\edef\entry{\thisrow{enidx}}\else\def\entry{nan}\fi
    \pgfkeyslet{/pgfplots/table/create col/next content}\entry
  }
]{ey_whole}{\enrawtable}
\pgfmathtruncatemacro{\enlast}{\the\encnt-1}
\begin{tikzpicture}
\begin{axis}[
  ultra/base,
  clip=true,
  xbar,
  bar width=7.4pt,
  width=3.85cm,
  height=4.3cm,
  xmode=log, xmin=0.18, xmax=35, log basis x=10,
  log origin x=infty,
  xtick={0.3,1,3,10}, xticklabels={0.3,1,3,10},
  y dir=reverse,
  ymin=-0.65, ymax=\enlast+0.65,
  ytick={0,...,\enlast},
  yticklabels/.expanded={\enylabels},
  yticklabel style={font=\scriptsize, align=right, inner sep=2.5pt},
  ymajorgrids=false,
  xmajorgrids=true,
  xlabel={Core Energy (Wh / audio-h $\equiv$ W / stream $\downarrow$, log)},
  xlabel style={font=\scriptsize},
  nodes near coords style={font=\tiny, inner sep=1.8pt, anchor=west},
]
\addplot[fill=okBlue!82, draw=okBlue!95!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=2]{\pgfplotspointmeta}\,W},
  point meta=explicit]
  table[x=watts, y=ey_ours, meta=watts]{\enrawtable};
\addplot[fill=okOrange!80, draw=okOrange!80!black, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=2]{\pgfplotspointmeta}\,W},
  point meta=explicit]
  table[x=watts, y=ey_stream, meta=watts]{\enrawtable};
\ifnum\ColdRowsWhole>0
\addplot[fill=okGrey!55, draw=okGrey!90!black, postaction={pattern=north east lines, pattern color=white!70}, bar shift=0pt,
  nodes near coords={\pgfmathprintnumber[fixed,fixed zerofill,precision=2]{\pgfplotspointmeta}\,W},
  point meta=explicit]
  table[x=watts, y=ey_whole, meta=watts]{\enrawtable};
\fi
\end{axis}
\end{tikzpicture}
