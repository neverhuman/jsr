#!/usr/bin/env bash
# R10 v4.2.1 release energy campaign driver (benchmark host, from ~/jsr-data/opponents/energy).
# R8 protocol unchanged (runs/r8/PREREG.md, R7 method): one pinned core (CPU 16), observer CPU 31, 60 s idle
# windows, QUIET.lock per arm, TIMING.lock only when no CI VM runs, one rerun of a core-unclean arm; the whole
# campaign holds the shared xbabe0 meter lease. Four arms on the full 730-utterance panel in three counterbalanced
# blocks: the R7/R8 anchors (RC2 live +LM, Zipformer INT8) and the isosyll live mode of the published v4.2.0
# x86_64-linux binary + model and the v4.2.1 glibc build (H318 endpoint H261 rerank, skip 3.75) + v4.2.1 model.
# usage: r10_campaign.sh
set -u
cd "$HOME/jsr-data/opponents/energy"
R=runs/r10/run-energy.sh
LOG=runs/r10/campaign.log
B=r10/bin
M9=r9/models
M=r10/models
declare -A cmdline=(
 [anchor730-live]="r7-live-lm @N panel730"
 [anchor730-zip]="r7-zipformer @N panel730"
 [v42-live-730]="r10-isosyll @N panel730 $B/v42/bin/isosyll $M9/v42/isosyll-model live"
 [v421-live-730]="r10-isosyll @N panel730 $B/v421/bin/isosyll $M/v421/isosyll-model live"
)
say() { echo "$(date -u +%FT%TZ) $*" | tee -a "$LOG"; }
exec 9>>/dev/shm/jsr-h172-meter.lease
say "waiting for the shared meter lease"
flock 9
say "meter lease held"
TL=$HOME/jsr-data/opponents/TIMING.lock
HAVE_TL=0
take_timing_lock() {
  HAVE_TL=0
  local vms; vms=$(systemctl list-units 'neverhuman-xbabe0-lan-*' --state=active --no-legend --plain 2>/dev/null | grep -c neverhuman-xbabe0-lan- || true)
  if [[ ${vms:-0} -eq 0 ]] && mkdir "$TL" 2>/dev/null; then
    printf 'r10-energy pid=%s start=%s until=%s\n' "$$" "$(date -u +%FT%TZ)" "$(date -u -d '+40 min' +%FT%TZ)" > "$TL/owner"
    HAVE_TL=1
    say "TIMING.lock taken (no CI VM running): $(bash $HOME/jsr-data/opponents/timing-quiet/timing-quiet-wait.sh 300)"
  else
    say "TIMING.lock not taken (CI VMs active=${vms:-?} or lock held); QUIET.lock only"
  fi
}
release_timing_lock() { if [[ $HAVE_TL == 1 ]]; then rm -f "$TL/owner"; rmdir "$TL" 2>/dev/null; HAVE_TL=0; fi; }
trap release_timing_lock EXIT
for b in 1 2 3; do
  case $b in
   1) order=(anchor730-live anchor730-zip v42-live-730 v421-live-730) ;;
   2) order=(v421-live-730 v42-live-730 anchor730-zip anchor730-live) ;;
   3) order=(v42-live-730 anchor730-live v421-live-730 anchor730-zip) ;;
  esac
  say "[b$b] BLOCK START order: ${order[*]}"
  for id in "${order[@]}"; do
    name="r10/b$b-r10-$id"
    if [[ -s runs/$name.json ]] && python3 -c "import json,sys;sys.exit(0 if json.load(open(sys.argv[1]))['clean_repetitions']['core']==1 else 1)" runs/$name.json; then say "[b$b] SKIP $id"; continue; fi
    for attempt in 1 2; do
      rm -rf "runs/out/$name"
      read -r -a args <<< "${cmdline[$id]//@N/$name}"
      while [[ -e $TL ]]; do say "[b$b] TIMING.lock present, waiting"; sleep 30; done
      take_timing_lock
      until bash $R quiet 45 "${args[@]}" >> runs/r10/arm-$b-$id.log 2>&1; do
        rc=$?; say "[b$b] quiet wrapper rc=$rc for $id, retry in 60 s"; sleep 60
        [[ -s runs/$name.json ]] && break
      done
      release_timing_lock
      clean=$(python3 - "runs/$name.json" <<'PY'
import json,sys
d=json.load(open(sys.argv[1]))
print(d.get("clean_repetitions",{}).get("core"), d.get("joules_per_audio_s"), d.get("cpu_s_per_audio_s"), ",".join(d.get("flags",[])))
PY
)
      say "[b$b] DONE $id attempt=$attempt core_clean/J/cpu/flags: $clean"
      if [[ ${clean%% *} == 1 ]]; then break; fi
      if [[ $attempt == 1 ]]; then
        say "[b$b] UNCLEAN core rep for $id; keeping it as .unclean and rerunning once"
        mv "runs/$name.json" "runs/$name.unclean.json"
        mv "runs/out/$name" "runs/out/$name.unclean"; mv "runs/series/b$b-r10-$id.json" "runs/series/b$b-r10-$id.unclean.json" 2>/dev/null
      fi
    done
  done
  say "[b$b] BLOCK DONE"
done
say "CAMPAIGN DONE"
